{{- /* Builds the 2-level category index in one pass over posts. Returns primaries and their subs sorted by name (case-insensitive); pages inside each sub are newest first: { name, key, count, latest, link, recent, subs: [ { name, key, count, link, pages } ] } recent is the primary's 2 newest posts across all subs: [ { page, sub } ]. Links come from the pages generated by scripts/generate-categories.go, so URLs never depend on matching Go's slugify with Hugo's urlize. */ -}} {{- $s := newScratch -}} {{- range $p := site.Pages -}} {{- with $p.Params.primary_category -}} {{- $key := . -}} {{- with $p.Params.secondary_category -}} {{- $key = printf "%s|%s" $key . -}} {{- end -}} {{- $s.SetInMap "link" $key $p.RelPermalink -}} {{- end -}} {{- end -}} {{- range $p := (where site.RegularPages "Section" "post").ByDate.Reverse -}} {{- $c := $p.Params.categories -}} {{- if ge (len $c) 2 -}} {{- $m := index $c 0 -}} {{- $sub := printf "%s|%s" $m (index $c 1) -}} {{- $s.SetInMap "mains" $m $m -}} {{- $s.SetInMap (printf "subs|%s" $m) $sub (index $c 1) -}} {{- $s.Add (printf "n|%s" $m) 1 -}} {{- $s.Add (printf "pages|%s" $sub) (slice $p) -}} {{- if lt ($s.Get (printf "rc|%s" $m) | default 0) 2 -}} {{- $s.Add (printf "recent|%s" $m) (slice (dict "page" $p "sub" (index $c 1))) -}} {{- $s.Add (printf "rc|%s" $m) 1 -}} {{- end -}} {{- if not ($s.Get (printf "latest|%s" $m)) -}} {{- $s.Set (printf "latest|%s" $m) $p.Date -}} {{- end -}} {{- end -}} {{- end -}} {{- $links := $s.Get "link" -}} {{- $result := slice -}} {{- range $m := $s.Get "mains" -}} {{- $subs := slice -}} {{- range $sub, $name := $s.Get (printf "subs|%s" $m) -}} {{- $pages := $s.Get (printf "pages|%s" $sub) -}} {{- $subs = $subs | append (dict "name" $name "key" (lower $name) "count" (len $pages) "link" (index $links $sub) "pages" $pages) -}} {{- end -}} {{- $result = $result | append (dict "name" $m "key" (lower $m) "count" ($s.Get (printf "n|%s" $m)) "latest" ($s.Get (printf "latest|%s" $m)) "link" (index $links $m) "recent" ($s.Get (printf "recent|%s" $m)) "subs" (sort $subs "key")) -}} {{- end -}} {{- return sort $result "key" -}}
