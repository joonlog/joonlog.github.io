{{- /* Builds the 2-level category index in one pass over posts. Returns a slice of primaries sorted by post count (desc): { name, count, latest, link, subs: [ { name, count, link, pages } ] } Links come from the pages generated by scripts/generate-categories.go, so URLs never depend on matching Go's slugify with Hugo's urlize. */ -}} {{- $s := newScratch -}} {{- range $p := site.Pages -}} {{- with $p.Params.primary_category -}} {{- $key := . -}} {{- with $p.Params.secondary_category -}} {{- $key = printf "%s|%s" $key . -}} {{- end -}} {{- $s.SetInMap "link" $key $p.RelPermalink -}} {{- end -}} {{- end -}} {{- range $p := (where site.RegularPages "Section" "post").ByDate.Reverse -}} {{- $c := $p.Params.categories -}} {{- if ge (len $c) 2 -}} {{- $m := index $c 0 -}} {{- $sub := printf "%s|%s" $m (index $c 1) -}} {{- $s.SetInMap "mains" $m $m -}} {{- $s.SetInMap (printf "subs|%s" $m) $sub (index $c 1) -}} {{- $s.Add (printf "n|%s" $m) 1 -}} {{- $s.Add (printf "pages|%s" $sub) (slice $p) -}} {{- if not ($s.Get (printf "latest|%s" $m)) -}} {{- $s.Set (printf "latest|%s" $m) $p.Date -}} {{- end -}} {{- end -}} {{- end -}} {{- $links := $s.Get "link" -}} {{- $result := slice -}} {{- range $m := $s.Get "mains" -}} {{- $subs := slice -}} {{- range $sub, $name := $s.Get (printf "subs|%s" $m) -}} {{- $pages := $s.Get (printf "pages|%s" $sub) -}} {{- $subs = $subs | append (dict "name" $name "count" (len $pages) "link" (index $links $sub) "pages" $pages) -}} {{- end -}} {{- $result = $result | append (dict "name" $m "count" ($s.Get (printf "n|%s" $m)) "latest" ($s.Get (printf "latest|%s" $m)) "link" (index $links $m) "subs" $subs) -}} {{- end -}} {{- return sort $result "count" "desc" -}}
